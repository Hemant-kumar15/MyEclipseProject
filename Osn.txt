*

for ((i=1; i<=6; i++))
do
  for ((j=1; j<=i; j++))
  do
    echo -n "*"
  done
  echo
done

-- Traffic light
 Input:
#!/bin/bash
echo "Enter light color (red / yellow / green):"
read color

case $color in
  red)
    echo "Stop"
    ;;
  yellow)
    echo "Wait"
    ;;
  green)
    echo "Go"
    ;;
  *)
    echo "Invalid color"
    ;;
esac

Ouput:
$ vi traffic.sh
$ chmod 754 traffic.sh
$ ./traffic.sh
Enter light color (red / yellow / green):
red
Stop
