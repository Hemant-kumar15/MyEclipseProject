class Pair<T, U> {
    T first;
    U second;

    Pair(T f, U s) {
        first = f;
        second = s;
    }
}

class Main {
    public static void main(String[] args) {
        Pair<String, Integer> p = new Pair<>("Amol", 20);
        System.out.println(p.first + " " + p.second);
    }
}
